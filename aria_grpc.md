# Adding gRPC to Aria — Internal AI Query Service

## Why gRPC, and why here specifically

Aria's current API surface is:
- **REST (Chi)** — used by the Next.js frontend for chat, feedback, conversations
- **SSE** — used by the frontend for streaming LLM tokens

Both are correct for their use cases. REST for request/response, SSE for server-push streaming.

gRPC belongs in a different layer: **internal service-to-service communication.**

Right now, if another service inside your infrastructure wants to query Aria's AI pipeline — a mobile backend, a webhook processor, a background job, a different microservice — it has to go through the public REST API. That means:
- JSON serialisation overhead on every call
- No strongly typed contract between services
- No bidirectional streaming (SSE is one-directional, server → client only)
- No native deadlines/cancellation propagated across services

gRPC solves all four. You add a gRPC server to the same Go binary — the REST API and gRPC server run simultaneously on different ports. The frontend still uses REST + SSE. Internal services use gRPC.

## What the gRPC service exposes

Two RPCs:

```protobuf
// proto/aria/v1/query.proto

syntax = "proto3";
package aria.v1;
option go_package = "github.com/Deonkar/aria/gen/aria/v1";

service AIQueryService {
  // Unary RPC — returns complete answer in one response
  // Used by: internal services that need a programmatic answer, not a stream
  rpc Query(QueryRequest) returns (QueryResponse);

  // Server-streaming RPC — streams tokens as they arrive from OpenAI
  // Used by: any client that wants real-time streaming (same as SSE but typed)
  rpc StreamQuery(QueryRequest) returns (stream QueryChunk);
}

message QueryRequest {
  string question    = 1;
  string agent_id    = 2;
  string agent_role  = 3;  // "agent" or "admin"
  string agent_name  = 4;
  string session_id  = 5;
  string timezone    = 6;
}

message QueryResponse {
  string answer          = 1;
  string generated_sql   = 2;
  int32  row_count       = 3;
  int32  execution_ms    = 4;
  bool   was_cached      = 5;
  bool   was_corrected   = 6;
  int32  tokens_input    = 7;
  int32  tokens_output   = 8;
}

message QueryChunk {
  enum ChunkType {
    CHUNK_TYPE_UNSPECIFIED = 0;
    CHUNK_TYPE_SQL         = 1;  // The generated SQL — arrives first
    CHUNK_TYPE_TOKEN       = 2;  // A streamed LLM token
    CHUNK_TYPE_DONE        = 3;  // Final metadata
    CHUNK_TYPE_ERROR       = 4;  // Error occurred
  }
  ChunkType type     = 1;
  string    text     = 2;  // Token text (CHUNK_TYPE_TOKEN) or SQL (CHUNK_TYPE_SQL)
  string    message_id = 3; // Populated in CHUNK_TYPE_DONE
  int32     row_count  = 4; // Populated in CHUNK_TYPE_DONE
  bool      cached     = 5; // Populated in CHUNK_TYPE_DONE
  string    error      = 6; // Populated in CHUNK_TYPE_ERROR
}
```

## Step 3 — Implement the gRPC server

```go
// internal/grpc/server.go

package grpcserver

import (
    "context"
    "io"

    ariav1 "github.com/Deonkar/aria/gen/aria/v1"
    "github.com/Deonkar/aria/internal/ai"
    "google.golang.org/grpc/codes"
    "google.golang.org/grpc/status"
)

// Server implements the generated AIQueryServiceServer interface
type Server struct {
    ariav1.UnimplementedAIQueryServiceServer  // embed for forward compatibility
    queryService *ai.QueryService
}

func NewServer(qs *ai.QueryService) *Server {
    return &Server{queryService: qs}
}

// Unary RPC — returns complete answer
func (s *Server) Query(ctx context.Context, req *ariav1.QueryRequest) (*ariav1.QueryResponse, error) {
    if req.AgentId == "" {
        return nil, status.Error(codes.InvalidArgument, "agent_id is required")
    }
    if req.Question == "" {
        return nil, status.Error(codes.InvalidArgument, "question is required")
    }

    result, err := s.queryService.Query(ctx, ai.QueryRequest{
        Question:  req.Question,
        AgentID:   req.AgentId,
        AgentRole: req.AgentRole,
        AgentName: req.AgentName,
        SessionID: req.SessionId,
        Timezone:  req.Timezone,
    })
    if err != nil {
        return nil, status.Errorf(codes.Internal, "query failed: %v", err)
    }

    return &ariav1.QueryResponse{
        Answer:        result.Answer,
        GeneratedSql:  result.SQL,
        RowCount:      int32(result.RowCount),
        ExecutionMs:   int32(result.ExecutionMs),
        WasCached:     result.WasCached,
        WasCorrected:  result.WasCorrected,
        TokensInput:   int32(result.TokensIn),
        TokensOutput:  int32(result.TokensOut),
    }, nil
}

// Server-streaming RPC — streams tokens
func (s *Server) StreamQuery(req *ariav1.QueryRequest, stream ariav1.AIQueryService_StreamQueryServer) error {
    ctx := stream.Context()

    if req.AgentId == "" {
        return status.Error(codes.InvalidArgument, "agent_id is required")
    }

    // Use a channel to receive tokens from the AI service
    tokenCh := make(chan ai.StreamChunk, 10)

    go func() {
        s.queryService.StreamQuery(ctx, ai.QueryRequest{
            Question:  req.Question,
            AgentID:   req.AgentId,
            AgentRole: req.AgentRole,
            SessionID: req.SessionId,
        }, tokenCh)
    }()

    for chunk := range tokenCh {
        var protoChunk *ariav1.QueryChunk

        switch chunk.Type {
        case ai.ChunkTypeSQL:
            protoChunk = &ariav1.QueryChunk{
                Type: ariav1.QueryChunk_CHUNK_TYPE_SQL,
                Text: chunk.Text,
            }
        case ai.ChunkTypeToken:
            protoChunk = &ariav1.QueryChunk{
                Type: ariav1.QueryChunk_CHUNK_TYPE_TOKEN,
                Text: chunk.Text,
            }
        case ai.ChunkTypeDone:
            protoChunk = &ariav1.QueryChunk{
                Type:      ariav1.QueryChunk_CHUNK_TYPE_DONE,
                MessageId: chunk.MessageID,
                RowCount:  int32(chunk.RowCount),
                Cached:    chunk.Cached,
            }
        case ai.ChunkTypeError:
            protoChunk = &ariav1.QueryChunk{
                Type:  ariav1.QueryChunk_CHUNK_TYPE_ERROR,
                Error: chunk.Error,
            }
        }

        if err := stream.Send(protoChunk); err != nil {
            // Client disconnected
            if err == io.EOF {
                return nil
            }
            return status.Errorf(codes.Internal, "stream send failed: %v", err)
        }
    }

    return nil
}
```

## Step 4 — Start gRPC server alongside HTTP server

```go
// cmd/api/main.go (additions)

import (
    "net"
    grpcserver "github.com/Deonkar/aria/internal/grpc"
    "google.golang.org/grpc"
    "google.golang.org/grpc/reflection"
    ariav1 "github.com/Deonkar/aria/gen/aria/v1"
)

func main() {
    cfg := config.Load()

    // ... existing setup: db, redis, openai, queryService ...

    // ── gRPC server ───────────────────────────────────────────────
    grpcSrv := grpc.NewServer(
        grpc.UnaryInterceptor(loggingUnaryInterceptor),    // log every RPC
        grpc.StreamInterceptor(loggingStreamInterceptor),  // log every stream
    )
    ariav1.RegisterAIQueryServiceServer(grpcSrv, grpcserver.NewServer(queryService))
    reflection.Register(grpcSrv) // enables grpcurl introspection in dev

    lis, err := net.Listen("tcp", ":9090")
    if err != nil {
        log.Fatal().Err(err).Msg("failed to listen on :9090")
    }

    // Start gRPC in background goroutine
    go func() {
        log.Info().Msg("gRPC server starting on :9090")
        if err := grpcSrv.Serve(lis); err != nil {
            log.Fatal().Err(err).Msg("gRPC server failed")
        }
    }()

    // ── HTTP server (existing) ────────────────────────────────────
    log.Info().Msg("HTTP server starting on :8080")
    http.ListenAndServe(":8080", router)
}
```

## Step 5 — Add gRPC interceptors (logging + recovery)

```go
// internal/grpc/interceptors.go

func loggingUnaryInterceptor(
    ctx context.Context,
    req interface{},
    info *grpc.UnaryServerInfo,
    handler grpc.UnaryHandler,
) (interface{}, error) {
    start := time.Now()
    resp, err := handler(ctx, req)

    log.Info().
        Str("method", info.FullMethod).
        Dur("duration_ms", time.Since(start)).
        Bool("success", err == nil).
        Msg("grpc unary")

    return resp, err
}

func recoveryInterceptor(
    ctx context.Context,
    req interface{},
    info *grpc.UnaryServerInfo,
    handler grpc.UnaryHandler,
) (resp interface{}, err error) {
    defer func() {
        if r := recover(); r != nil {
            log.Error().Interface("panic", r).Msg("grpc panic recovered")
            err = status.Errorf(codes.Internal, "internal server error")
        }
    }()
    return handler(ctx, req)
}
```

## What this adds to your resume

**Before:** "Built AI assistant with REST API and SSE streaming in Go"

**After:** "Exposed Aria's AI query pipeline as a gRPC service (proto3, server-streaming RPC for token delivery) alongside the existing REST API — gRPC on :9090 for internal service-to-service calls, REST + SSE on :8080 for the frontend. Both run in the same Go binary."

The interviewer question this unlocks: *"Why gRPC alongside REST instead of just REST?"* — and your answer is: REST is optimal for browser clients (SSE, JSON, CORS), gRPC is optimal for internal service calls (binary protocol, typed contract, bidirectional streaming, deadline propagation). Different clients, different protocols, same underlying query service.
